### 2-3. 격자 분포 — 어느 칸이 최악인가

## C-2. run.ts — 변형 3종 실행 + 복제 대 원본 동치 확인

```typescript
// 사용: vite-node run.ts <profile> <strategy> <n> <outJson> [equivN]
// 변형 3종(0 기준 / 1 L1 / 2 L1+L3)을 같은 시드로 돌려 판별 기록을 JSON으로 남긴다. 복제 기준 대 원본 최종 상태 동치를 equivN판 확인.
import fs from 'node:fs';
import { simulateLife as original } from '<저장소>/scripts/simulateLib';
import { simulateLife } from './clone/scripts/simulateLib';
import { cx, cxSet } from './clone/src/core/economy';
import type { ProfileId } from './clone/src/core/types';

const [profile, strategy, nS, outPath, equivS] = process.argv.slice(2);
const n = Number(nS);
const equivN = Number(equivS ?? 20);
const seedOf = (i: number) => (i * 2654435761) >>> 0;
const VARIANTS: Array<[string, number, boolean]> = [
  ['base', 0, false],
  ['L1', 1200, false],
  ['L1L3', 1200, true],
];
const out: Record<string, unknown[]> = {};
let equivChecked = 0;
let equivBad = 0;
for (const [name, floor, pension] of VARIANTS) {
  const rows: unknown[] = [];
  for (let i = 1; i <= n; i++) {
    const seed = seedOf(i);
    cxSet(floor, pension);
    const life = simulateLife(seed, profile as ProfileId, strategy as 'auto' | 'random');
    if (name === 'base' && i <= equivN) {
      const o = original(seed, profile as ProfileId, strategy as 'auto' | 'random');
      equivChecked++;
      if (JSON.stringify(o.final) !== JSON.stringify(life.final)) equivBad++;
    }
    const f = life.final;
    rows.push({
      seed,
      tier: f.character!.parentWealth,
      death: life.deathAge,
      net: f.assets?.net ?? null,
      debt: f.assets?.debt ?? null,
      dwelling: f.assets?.dwelling ?? null,
      flows: cx.flows.map((x) => ({ ...x })),
      hist: f.history.filter((h) => h.age >= 20 && h.eventId !== undefined).map((h) => [h.age, h.eventId, h.kind]),
    });
  }
  out[name] = rows;
}
fs.writeFileSync(outPath, JSON.stringify({ profile, strategy, n, equivChecked, equivBad, out }));
console.log(JSON.stringify({ profile, strategy, n, equivChecked, equivBad }));
```

```typescript
P('\n## D6. 음수 종료 판의 경로');
function seqOf(r: Row) { const seq: Array<{ age: number; ph: string; net: number }> = []; for (const f of r.flows) { seq.push({ age: f.age, ph: 'ev', net: f.before }); seq.push({ age: f.age, ph: 'fl', net: f.after }); } return seq; }
function stretchStart(r: Row) {
  const seq = seqOf(r); let k = -1;
  for (let i = 0; i < seq.length; i++) if (seq[i].net < 0 && (i === 0 || seq[i - 1].net >= 0)) k = i;
  return k < 0 ? null : { i: k, ...seq[k] };
}
const ageBand = (a: number) => (a < 30 ? '20대' : a < 40 ? '30대' : a < 50 ? '40대' : a < 60 ? '50대' : a < 65 ? '60~64' : a < 70 ? '65~69' : a < 80 ? '70대' : '80+');
const ABANDS = ['20대', '30대', '40대', '50대', '60~64', '65~69', '70대', '80+'];
for (const p of ['millennial', 'industrial', 'alpha'] as const) for (const s of STRATS) {
  const base = adultOf(data[p][s].rows.base); const neg = base.filter((r) => r.net! < 0);
  P(`\n### ${p} ${s}: 기준 음수 종료 ${neg.length}판`);
  // 마지막 음수 구간이 시작된 나이대와 원인(사건·상속 / 해 현금흐름)
  const cnt: Record<string, number> = {}; const cause: Record<string, number> = {}; const evIds: Record<string, number> = {};
  const flowAtCross: number[] = []; const evDelta: number[] = [];
  for (const r of neg) {
    const st = stretchStart(r); if (!st) continue;
    cnt[ageBand(st.age)] = (cnt[ageBand(st.age)] ?? 0) + 1;
    const key = `${ageBand(st.age)}/${st.ph === 'ev' ? '사건·상속' : '해 현금흐름'}`; cause[key] = (cause[key] ?? 0) + 1;
    if (st.ph === 'ev') { for (const h of r.hist) if (h[0] === st.age) evIds[h[1]] = (evIds[h[1]] ?? 0) + 1; }
    const f = at(r, st.age); if (f) { if (st.ph === 'fl') flowAtCross.push(f.total); else { const prev = at(r, st.age - 1); evDelta.push(f.before - (prev ? prev.after : f.before)); } }
  }
  P('음수 구간 시작 나이대: ' + ABANDS.map((b) => `${b} ${cnt[b] ?? 0}`).join(' · '));
  P('시작 나이대 / 시작 수단: ' + Object.entries(cause).sort().map(([k, v]) => `${k} ${v}`).join(' · '));
  P(`해 현금흐름으로 시작한 판의 그 해 total 중앙값 ${fmt(med(flowAtCross))} (n=${flowAtCross.length}) · 사건·상속으로 시작한 판의 그 해 사건 순변화 중앙값 ${fmt(med(evDelta))} (n=${evDelta.length})`);
  P('사건·상속으로 시작한 판의 그 나이 연표 이벤트 상위: ' + Object.entries(evIds).sort((a, b) => b[1] - a[1]).slice(0, 10).map(([k, v]) => `${k}(${v})`).join(', '));
  // 시작 시점 부채가 그 뒤 얼마나 커졌나, 끝 부채
  P(`끝 순자산 중앙값 ${fmt(med(neg.map((r) => r.net!)))} · 끝 부채 중앙값 ${fmt(med(neg.map((r) => r.debt!)))} · 끝 나이 중앙값 ${fmt(med(neg.map((r) => r.death)))} · 끝 주거 ${['rent', 'jeonse', 'own', 'family', 'care'].map((d) => d + ' ' + neg.filter((r) => r.dwelling === d).length).join(' · ')}`);
  P(`출생 재력 분포(1~5): ${[1, 2, 3, 4, 5].map((t) => neg.filter((r) => r.tier === t).length).join('/')} · 65세 이상 생존 중 음수 구간 시작이 65세 이후인 판: ${neg.filter((r) => { const st = stretchStart(r); return st && st.age >= 65; }).length}`);
  // 같은 판이 L1·L1L3에서 어떻게 바뀌었나
  for (const v of ['L1', 'L1L3'] as const) {
    const idx = new Map(data[p][s].rows[v].map((r) => [r.seed, r]));
    let changed = 0; let toPos = 0; let toF = 0; let toZero = 0; let stillNeg = 0; let diverge65Plus = 0; let divergeBelow65 = 0; const dvBase: number[] = []; const dvNet: string[] = []; const dvCat: Record<string, number> = {};
    for (const r of neg) {
      const o = idx.get(r.seed)!;
      if (o.net !== r.net) changed++;
      if (o.net! > 0) toPos++; if (o.net === F) toF++; if (o.net === 0) toZero++; if (o.net! < 0) stillNeg++;
      // 처음으로 연초(before)가 달라지는 나이 (사건 이후 같은 해에 이미 달라지면 그 해의 이전 해 흐름 차)
      let dv: any = null;
      for (let i = 0; i < r.flows.length && i < o.flows.length; i++) if (r.flows[i].before !== o.flows[i].before || r.flows[i].after !== o.flows[i].after) { dv = [r.flows[i], o.flows[i]]; break; }
      if (dv) {
        if (dv[0].age >= 65) diverge65Plus++; else divergeBelow65++;
        dvBase.push(dv[0].before);
        const b = dv[0].before; const cat = b < 0 ? 'before<0' : b === 0 ? 'before=0' : b <= F ? '(0,F]' : 'before>F';
        dvCat[cat] = (dvCat[cat] ?? 0) + 1;
        dvNet.push(`${dv[0].age}`);
      }
    }
    P(`- ${v}: 바뀐 판 ${changed}/${neg.length} · 양수로 마침 ${toPos} · 정확히 F ${toF} · 정확히 0 ${toZero} · 여전히 음수 ${stillNeg} · 처음 갈라지는 해가 65세 이상 ${diverge65Plus} / 65세 미만 ${divergeBelow65} · 갈라지는 해의 기준 연초 순자산: ${Object.entries(dvCat).map(([k, x]) => `${k} ${x}`).join(', ')} · 갈라지는 나이 중앙값 ${fmt(med(dvNet.map(Number)))}`);
  }
}
```
